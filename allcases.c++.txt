program AllCasesTest {
    declare x, y;
    x := 0;
    y := 10;

    switchcase
        when x = 0: { y := y + 1; x := x + 1 }
        when x = 1: { y := y - 1 }
        default: { y := 0; print y };

    
    forcase x = 10
        when x < 5: { x := x + 1; print x };

    whilecase
        when y > 0: { y := y - 2; x := x + 2 }
        default: { print x };

    incase
        when x >= y: { print x; y := y + 10 };

    untilcase
        when x < 50: { x := x + 5; print x }
        until x >= 50
}
